% eq01_rrf: Reciprocal rank fusion
\[

\]

% eq02_gam: Fusion as a GAM (EBM generalizes RRF)
\[
s(d)=\sum_{r} f_r\big(\mathrm{rank}_r(d)\big)+\sum_{r<r'} f_{rr'}\big(\mathrm{rank}_r(d),\mathrm{rank}_{r'}(d)\big)
\]

% eq03_calibration: RSA calibration
\[
p_C(x)=\sigma\big(a'_C\,F_C(x)+c_C\big)
\]

% eq04_composition: RSA composition
\[
S_Q(x)=\sum_{C\in Q^+}\log p_C(x)+\sum_{C\in Q^-}\log\big(1-p_C(x)\big)
\]

% eq05_popcount: RSA popcount kernel
\[
P_C(x)=a_C\,n_+(x)+\Delta_C\sum_{t=0}^{3}2^t\,\mathrm{popcnt}\big(B_x\wedge M_{Ct}\big)
\]

% eq08_sae: Sparse autoencoder

% eq09_rq: Residual quantization for semantic IDs

r_0=e(x),\quad c_\ell=\arg\min_k\lVert r_{\ell-1}-C_\ell[k]\rVert,\quad r_\ell=r_{\ell-1}-C_\ell[c_\ell],\quad \mathrm{ID}(x)=(c_1,\dots,c_L)
